# SP3 展墙 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the square, photographer page and album walls, wired to the SP2 viewer with position and focus restore.

**Architecture:** A small Go change adds `previews` to every card and makes the user wall id-scoped (`user:{id}`). The frontend adds `web/src/walls/` (pure layout/cache/restore modules plus card and wall components) and three lazy routes. The viewer's exit path maps the wall value to its route.

**Tech Stack:** Go (chi, SQLite), Vue 3.5 + vue-router 4.6, TypeScript, vitest + happy-dom, Playwright.

**Spec:** `docs/superpowers/specs/2026-10-17-sp3-walls-design.md` (parent: `docs/superpowers/specs/2026-10-13-markr-redesign-design.md` §4, §7, §10)

## Global Constraints

- No new npm or Go dependencies.
- Visual: walls use `--bg-wall`; no cards, shadows, hover zoom; radius 0; one accent `--accent` for focus (2px outline, 4px offset).
- Copy: Chinese, photographer's language, e.g. 「12 张」「载入中」「重试」. No marketing text. Visitor UI never says 「合集」.
- Accessibility: text contrast 4.5:1, visible focus, touch targets ≥ 44px, nothing hover-only.
- Bundle: `/c/:id` critical path ≤ 80 KB gz (existing); `/` critical path ≤ 80 KB gz (new). Wall code must not be statically imported by `CollectionRoute.vue`.
- Storage access always in try/catch; failures silent.
- Commits end with `Co-Authored-By: Claude Opus 5.5 <noreply@anthropic.com>`.
- Verification command (frontend): `cd web && npx vitest run && npx vue-tsc --noEmit && npm run build && npm run check:bundle`. Go: `go test ./internal/api/...` (MinIO not needed for these packages; if a test needs it see `.superpowers/HANDOFF.md`).

## Review Focus

1. Returning to the square after paging deep: the cached cards render first and the background refresh must not reorder or drop them, or the restore target jumps away. Tested in Task 5 (`SquareRoute` refresh test).
2. A collection whose cover image 404s or has `cover: null` must still show a sized placeholder with its title, never a zero-height hole. Tested in Task 4 (`WallCard`) and Task 6 (e2e broken cover).
3. Sweep on a card with 0 or 1 previews must not throw or flicker; with `previews: []` the cover stays. Tested in Task 3 (`sweepIndex`) and Task 4.
4. Legacy `wall=user` URLs (already shared or in history) must keep working and come back as `user:{owner}`. Tested in Task 1.
5. Narrow phones (360px) must not scroll horizontally even with panorama covers and long titles. Tested in Task 6 at 360/390/768/1440.

```ts
// layout.ts — pure placement for the salon wall (spec §4.1). No DOM.
export function hash(id: string): number {
  let h = 0x811c9dc5
  for (let i = 0; i < id.length; i++) {
    h ^= id.charCodeAt(i)
    h = Math.imul(h, 0x01000193)
  }
  return h >>> 0
}

export function columnsFor(width: number): 1 | 2 | 3 {
  return width < 600 ? 1 : width < 1100 ? 2 : 3
}

export interface Placement { column: number; widthPct: number; align: 'start' | 'center' | 'end' }
export interface LayoutItem { id: string; aspect: number; count: number; featured: boolean }

const ALIGNS = ['start', 'center', 'end'] as const
const MAX_PANO = 2.4

export function salon(items: readonly LayoutItem[], columns: number): Placement[] {
  const cols = Math.max(1, Math.floor(columns))
  const heights = new Array<number>(cols).fill(0)
  const min = cols === 1 ? 0.85 : 0.7
  return items.map((it) => {
    const h = hash(it.id)
    const aspect = Number.isFinite(it.aspect) && it.aspect > 0 ? it.aspect : 1.5
    let w = 0.7 + ((h % 1000) / 1000) * 0.3
    if (aspect < 1) w -= 0.08
    if (aspect > 2) w += 0.1
    if (it.count >= 12) w += 0.05
    if (it.featured) w = 1
    w = Math.min(1, Math.max(min, w))
    let column = 0
    for (let c = 1; c < cols; c++) if (heights[c]! < heights[column]!) column = c
    heights[column]! += w / Math.min(aspect, MAX_PANO) + 0.15 // + caption/gap allowance in column widths
    const widthPct = Math.round(w * 1000) / 10
    return { column, widthPct, align: widthPct >= 100 ? 'center' : ALIGNS[(h >>> 10) % 3]! }
  })
}

export function sweepIndex(x: number, width: number, n: number): number {
  if (n <= 0) return -1
  if (width <= 0) return 0
  return Math.min(n - 1, Math.max(0, Math.floor((x / width) * n)))
}
```

```ts
// wallCache.ts — per-tab wall snapshot so a return renders instantly (spec §6).
export const TTL_MS = 30 * 60 * 1000
const key = (wall: string) => `markr:wallCache:${wall}`

export function save<T>(wall: string, data: T, now = Date.now()): void {
  try {
    sessionStorage.setItem(key(wall), JSON.stringify({ savedAt: now, data }))
  } catch {
    // ignore
  }
}

export function load<T>(wall: string, now = Date.now()): T | null {
  try {
    const raw = sessionStorage.getItem(key(wall))
    if (!raw) return null
    const v = JSON.parse(raw) as { savedAt?: unknown; data?: T }
    if (typeof v.savedAt !== 'number' || now - v.savedAt > TTL_MS || v.data === undefined) {
      sessionStorage.removeItem(key(wall))
      return null
    }
    return v.data
  } catch {
    return null
  }
}
```

```ts
import { recall } from '../viewer/lastViewed'

export function restoreFocus(root: HTMLElement, wall: string): boolean {
  const id = recall(wall)
  if (!id) return false
  try {
    sessionStorage.removeItem(`markr:lastViewed:${wall}`)
  } catch {
    // ignore
  }
  const el = Array.from(root.querySelectorAll<HTMLElement>('[data-card-id]')).find((e) => e.dataset.cardId === id)
  if (!el) return false
  el.scrollIntoView({ block: 'center' })
  el.focus({ preventScroll: true })
  return true
}
```

```ts
// wallRoute.ts — the route a wall value lives at (spec §7).
export function wallRoute(wall: string | null, ownerId: string): string {
  if (wall === 'square') return '/'
  if (wall?.startsWith('album:') && wall.length > 6) return `/a/${encodeURIComponent(wall.slice(6))}`

}
```

```vue
  if (props.aspect) return props.aspect
```

```ts
const hover = typeof matchMedia === 'function' && matchMedia('(hover: hover)').matches
const shown = ref(-1) // -1 = cover
function onMove(e: PointerEvent) {
```
